<script>
  var $ = function (s) { return document.querySelector(s); };

  function to12h(v) {
    if (!v) return v;
    var p = v.split(":"), h = +p[0], ap = h >= 12 ? "PM" : "AM";
    return ((h % 12) || 12) + ":" + p[1] + " " + ap;
  }
  function toSheetDate(v) { var p = v.split("-"); return p[1] + "/" + p[2] + "/" + p[0]; }
  function todayStr() {
    var d = new Date(), m = ("0" + (d.getMonth() + 1)).slice(-2), day = ("0" + d.getDate()).slice(-2);
    return d.getFullYear() + "-" + m + "-" + day;
  }

  /* Builds all behavior for one form.
     cfg: { form, prefix, sessionField, buttonText, successText, waitText, serverFn, buildPayload(get, people) }
     sessionField is the data-f name of the AM/PM select ("ampm" on pick-up, "session" on drop-off).
     The individuals list depends on BOTH the route and the AM/PM run. */
  function setupForm(cfg) {
    var form = cfg.form;
    var f = function (name) { return form.querySelector('[data-f="' + name + '"]'); };
    var peopleEl = f("people"), submitBtn = f("submit"), statusEl = f("status"), msgEl = f("msg");
    var rosterBlocked = false, rosterReq = 0;

    function selectedPeople() {
      var sel = form.querySelector(".individual");
      return sel && sel.value ? [sel.value] : [];
    }

    function refresh() {
      var ok = form.checkValidity() && !rosterBlocked;
      submitBtn.disabled = !ok;
      statusEl.textContent = ok ? "Ready to submit." : cfg.waitText;
    }

    function renderPeople() {
      var code = f("route").value, session = f(cfg.sessionField).value, req = ++rosterReq;
      peopleEl.innerHTML = "";
      if (!code || !session) {
        rosterBlocked = false;
        peopleEl.innerHTML = '<p class="empty">Select a route and AM/PM to see its individuals.</p>';
        refresh();
        return;
      }
      rosterBlocked = true;
      peopleEl.innerHTML = '<p class="empty">Loading individuals...</p>';
      refresh();
      google.script.run
        .withSuccessHandler(function (names) {
          if (req !== rosterReq) return;
          rosterBlocked = false;
          peopleEl.innerHTML = "";
          if (!names || !names.length) {
            peopleEl.innerHTML = '<p class="empty">No individuals are listed for this route and run yet.</p>';
            refresh();
            return;
          }
          var selId = cfg.prefix + "_individual";
          var lab = document.createElement("label");
          lab.setAttribute("for", selId);
          lab.textContent = "Individual";
          var sel = document.createElement("select");
          sel.id = selId; sel.className = "individual"; sel.required = true;
          var first = document.createElement("option");
          first.value = ""; first.textContent = "Select an individual";
          sel.appendChild(first);
          names.forEach(function (n) {
            var o = document.createElement("option");
            o.value = n; o.textContent = n;
            sel.appendChild(o);
          });
          peopleEl.appendChild(lab);
          peopleEl.appendChild(sel);
          refresh();
        })
        .withFailureHandler(function (err) {
          if (req !== rosterReq) return;
          peopleEl.innerHTML = '<p class="empty">Could not load individuals: ' +
            (err && err.message ? err.message : err) + ' Pick the route or AM/PM again to retry.</p>';
          refresh();
        })
        .getRouteMembers(code, session);
    }

    f("route").addEventListener("change", renderPeople);
    f(cfg.sessionField).addEventListener("change", renderPeople);
    form.addEventListener("input", refresh);
    form.addEventListener("change", refresh);
    form.addEventListener("focusout", function (e) { e.target.classList.add("touched"); });
    if (cfg.onSetup) cfg.onSetup(f, refresh);

    form.addEventListener("submit", function (e) {
      e.preventDefault();
      if (!form.checkValidity()) { form.reportValidity(); return; }
      var payload = cfg.buildPayload(function (n) { return f(n).value; }, selectedPeople());
      submitBtn.disabled = true; submitBtn.textContent = "Submitting..."; msgEl.className = "msg";
      google.script.run
        .withSuccessHandler(function (r) {
          msgEl.className = "msg ok";
          msgEl.textContent = cfg.successText + r.file + ".";
          form.reset(); renderPeople();
          f("serviceDate").value = todayStr();
          if (cfg.onSuccess) cfg.onSuccess(payload);
          submitBtn.textContent = cfg.buttonText; refresh();
          window.scrollTo({ top: document.body.scrollHeight, behavior: "smooth" });
        })
        .withFailureHandler(function (err) {
          msgEl.className = "msg err";
          msgEl.textContent = "Not submitted: " + (err && err.message ? err.message : err) + " Your entries are still here. Fix the problem and submit again.";
          submitBtn.textContent = cfg.buttonText; refresh();
        })[cfg.serverFn](payload);
    });

    f("serviceDate").value = todayStr();
    return { f: f, refresh: refresh };
  }

  /* ---- Pick-up ---- */
  var pick = setupForm({
    form: $("#fPick"), prefix: "p", serverFn: "submitPickUp", sessionField: "ampm",
    buttonText: "Submit pick-up log",
    waitText: "Complete every required field to enable submit.",
    successText: "Pick-up log submitted and filed as ",
    onSuccess: function (payload) { lockLicense(payload.license); },
    buildPayload: function (get, people) {
      return {
        route: get("route"),
        serviceDate: toSheetDate(get("serviceDate")),
        license: get("license").trim(),
        individuals: people,
        ampm: get("ampm"),
        pickup: to12h(get("pickup")),
        driverName: get("driverName").trim(),
        driverSignature: get("driverSignature").trim()
      };
    }
  });

  /* Driver's license: strip spaces/dashes, capitalize, 1 letter then 8 numbers, 9-character counter.
     After the first successful pick-up the number is kept in memory for this page only. */
  var savedLicense = "";
  var lic = $("#p_license"), licCountEl = $("#licCount");
  function updateLicCount() {
    var n = lic.value.length;
    licCountEl.textContent = n === 9 ? "9 of 9 characters - looks good!" : n + " of 9 characters typed";
    licCountEl.className = "count" + (n === 9 ? " good" : "");
  }
  lic.addEventListener("input", function () {
    var raw = this.value.toUpperCase(), clean = "";
    for (var i = 0; i < raw.length && clean.length < 9; i++) {
      var ch = raw.charAt(i);
      if (clean.length === 0) { if (/[A-Z]/.test(ch)) clean += ch; }
      else if (/[0-9]/.test(ch)) clean += ch;
    }
    if (clean !== this.value) this.value = clean;
    updateLicCount();
  });
  // form.reset() clears the box, so put the saved license back
  $("#fPick").addEventListener("reset", function () {
    setTimeout(function () {
      if (savedLicense) lic.value = savedLicense;
      updateLicCount();
      pick.refresh();
    }, 0);
  });
  function lockLicense(value) {
    savedLicense = value;
    $("#licMasked").textContent = "\u2022\u2022\u2022\u2022\u2022" + value.slice(-4);
    $("#licField").classList.add("hidden");
    $("#licHelp").classList.add("hidden");
    $("#licSaved").classList.remove("hidden");
    $("#keepOpen").classList.remove("hidden");
    window.onbeforeunload = function (e) { e.preventDefault(); e.returnValue = ""; return ""; };
  }
  $("#licSwitch").addEventListener("click", function () {
    savedLicense = ""; lic.value = "";
    $("#licField").classList.remove("hidden");
    $("#licHelp").classList.remove("hidden");
    $("#licSaved").classList.add("hidden");
    $("#keepOpen").classList.add("hidden");
    window.onbeforeunload = null;
    updateLicCount(); pick.refresh(); lic.focus();
  });

  /* ---- Drop-off ---- */
  var drop = setupForm({
    form: $("#fDrop"), prefix: "d", serverFn: "submitDropOff", sessionField: "session",
    buttonText: "Submit drop-off log",
    waitText: "Complete every field to enable submit.",
    successText: "Drop-off log submitted and filed as ",
    buildPayload: function (get, people) {
      return {
        route: get("route"),
        serviceDate: toSheetDate(get("serviceDate")),
        individuals: people,
        dropoff: to12h(get("dropoff")),
        session: get("session"),
        signature: get("signature").trim()
      };
    }
  });

  /* ---- Chooser / navigation ---- */
  function show(which) {
    $("#chooser").classList.toggle("hidden", which !== "choose");
    $("#pickWrap").classList.toggle("hidden", which !== "pick");
    $("#dropWrap").classList.toggle("hidden", which !== "drop");
    window.scrollTo(0, 0);
  }
  $("#choosePick").addEventListener("click", function () { show("pick"); });
  $("#chooseDrop").addEventListener("click", function () { show("drop"); });
  Array.prototype.forEach.call(document.querySelectorAll("[data-back]"), function (b) {
    b.addEventListener("click", function () { show("choose"); });
  });

  /* ---- Load routes once for both forms ---- */
  google.script.run.withSuccessHandler(function (routes) {
    [pick, drop].forEach(function (x) {
      var sel = x.f("route");
      routes.forEach(function (h) {
        var o = document.createElement("option"); o.value = h.code; o.textContent = h.label; sel.appendChild(o);
      });
      x.refresh();
    });
  }).getRouteList();
</script>
